\begin{figure}[htbp]
    \centering
    \begin{tikzpicture}[
        >=stealth,
        line width=0.9pt,
        every node/.style={font=\large}
    ]
    % Manifold M_r as a smooth curve
    \draw[thick] plot[smooth, domain=-2.8:2.8, samples=60]
        ({\x}, {0.22*\x*\x - 1});
    \node at (3, 0.9) {$\mathcal{M}_r$};

    % Tangent line at X^(i)
    \draw[dashed] (-2.8, -1) -- (3, -1);
    \node[right] at (3, -1) {$\mathcal{T}_{X^{(i)}}(\mathcal{M}_r)$};

    % Point X^(i)
    \coordinate (X) at (0, -1);
    \fill (X) circle (2.5pt);
    \node[below right] at (X) {$X^{(i)}$};

    % -grad Phi(X^(i)): full Euclidean gradient direction
    \coordinate (G) at (-1.4, 1);
    \draw[->] (X) -- (G);
    \node[above left] at (G) {$-\nabla\Phi(X^{(i)})$};

    % Dotted projection from -grad Phi tip down onto the tangent line
    \draw[dotted] (G) -- (-1.4, -1);

    % -P G^(i): projection of -grad Phi onto T_{X^(i)} M_r
    \coordinate (PG) at (-1.4, -1);
    \draw[->] (X) -- (PG);
    \node[below] at (-0.7, -1) {$-\mathcal{P}G^{(i)}$};

    % D^(i): CG search direction, generated by the recurrence
    \coordinate (D) at (1.8, 1.4);
    \draw[->] (X) -- (D);
    \node[above right] at (D) {$D^{(i)}$};

    \end{tikzpicture}
    \caption{The search direction $D^{(i)}$ generated by
    the DLRA-CG may disagree with both the negative Euclidean gradient $-\nabla\Phi(X^{(i)})$ and its projection $-\mathcal{P}G^{(i)} = -\mathcal{P}_{X^{(i)}}(\nabla\Phi(X^{(i)}))$ onto the
    tangent space $\mathcal{T}_{X^{(i)}}(\mathcal{M}_r)$.}
    \label{fig:cg_direction_disagreement}
